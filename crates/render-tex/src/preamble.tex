% Page flow: a paragraph never leaves a single line behind or ahead of a
% page break, so a two-line antiphon or verse always stays whole. Pages keep
% their natural height rather than stretching the leading to fill.

\displaywidowpenalty=10000

% Running heads: the hour on the left page, the day on the right, in muted
% small caps; the folio centred below. The opening page carries only the
% folio. \officehour and \officeday are defined per booklet.
\newcommand{\runninghead}[1]{{\footnotesize\color{mutedgray}\scshape #1}}
\fancypagestyle{office}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \fancyhead[LE]{\runninghead{\officehour}}%
  \fancyhead[RO]{\runninghead{\officeday}}%
  \fancyfoot[C]{{\small\color{mutedgray}\thepage}}%
}
\fancypagestyle{plain}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \fancyfoot[C]{{\small\color{mutedgray}\thepage}}%
}
\pagestyle{office}

% Gilt hairline with a fleuron at its centre, framing the title. The closing
% \null keeps TeX from discarding the right-hand leaders as trailing glue.
\newcommand{\ornamentrule}{%
  \noindent{\color{goldline}\leaders\hrule height 2.9pt depth -2.5pt\hfill}%
  \enspace{\large\color{ornamentgold}❦}\enspace%
  {\color{goldline}\leaders\hrule height 2.9pt depth -2.5pt\hfill}\null\par%
}

% Title: \officetitle{hour}{date}{feast}{season}{color name}{color key}.
% The feast (#3) and the season line (#4, #5) may be empty.
\newcommand{\officetitle}[6]{%
  \thispagestyle{plain}%
  \begin{center}%
    \ornamentrule\vspace{10pt}%
    {\small\color{mutedgray}\scshape The Order for}\\[3pt]%
    {\LARGE\scshape #1}\\[8pt]%
    \if\relax\detokenize{#3}\relax\else{\large\itshape #3}\\[4pt]\fi%
    {\normalsize #2}\\[5pt]%
    \if\relax\detokenize{#4#5}\relax\else%
      {\small\color{mutedgray}#4%
        \if\relax\detokenize{#4}\relax\else\if\relax\detokenize{#5}\relax\else\enspace$\cdot$\enspace\fi\fi%
        \if\relax\detokenize{#5}\relax\else{\color{lit-#6}\rule[0.05em]{0.55em}{0.55em}}\,\,#5\fi}\\[2pt]%
    \fi%
    \vspace{6pt}\ornamentrule%
  \end{center}%
  \vspace{4pt}%
}

% Rubrics are smaller red roman; quoted prayer remains ordinary black roman.
\newcommand{\rubric}[1]{{\color{rubricred}\small\normalfont #1}}
% A prayer quoted inside a rubric remains ordinary black text.
\newcommand{\rubricprayed}[1]{{\color{black}\normalfont #1}}
% Scripture references share the rubric treatment without becoming instructions.
\newcommand{\scriptureref}[1]{{\color{rubricred}\small\normalfont #1}}

% Gilded two-line initials. All call sites are semantic spoken openings, never
% private/secret prayer text. Only the initial itself is gilt: the rest of the
% opening word stays ordinary prose, as ::first-letter does on the web hour.
\renewcommand{\LettrineTextFont}{\normalfont}
\newif\ifcapopen
\newcommand{\dropcap}[2]{\global\capopentrue\lettrine[lines=2,lhang=0.1,nindent=0.1em]{\color{ornamentgold}#1}{#2}}
% Call after the \par that ends a paragraph which may open with \dropcap. An
% initial is two lines deep, so when its paragraph set only one line the
% next paragraph would run into it: open just enough space to clear it.
\newcommand{\capclear}{%
  \ifcapopen\global\capopenfalse\ifnum\prevgraf<2 \vspace{0.55\baselineskip}\fi\fi%
}

% Section heading: centered small caps with open air, as the web hour centers
% each transition in tracked small caps. Never stranded at the foot of a page.
\newcommand{\sectionheading}[1]{%
  \Needspace{6\baselineskip}\bigskip\begin{center}{\scshape #1}\end{center}\nopagebreak\smallskip\noindent%
}

% Mid-canticle section break (e.g. the two-part structure of Deut 32):
% centered muted italic, as the web canticle-section line.
\newcommand{\canticlesection}[1]{%
  \Needspace{4\baselineskip}\smallskip\begin{center}{\small\color{mutedgray}\itshape #1}\end{center}\nopagebreak\smallskip\noindent%
}

% Psalm/canticle label: the number in body-size small caps, the Latin incipit in italic
% beside it, as a printed diurnal sets them. #2 is empty when no incipit is
% recorded, in which case the label stands alone. Kept with its first verses.
\newcommand{\psalmlabel}[2]{%
  \Needspace{5\baselineskip}%
  \begin{center}{\color{mutedgray}\scshape #1}%
  \if\relax\detokenize{#2}\relax\else{\color{mutedgray}\itshape\enspace$\cdot$\enspace #2}\fi%
  \end{center}\nopagebreak\smallskip%
}

% Hymn or anthem title (usually its Latin incipit): muted italic, as the
% incipit beside a psalm label. Kept with its first stanza.
\newcommand{\hymnlabel}[1]{%
  \Needspace{5\baselineskip}%
  \begin{center}{\color{mutedgray}\itshape #1}\end{center}\nopagebreak\smallskip%
}

% Pointing mediant: muted and raised to the optical middle of the line, as the
% web verse renderer lifts the asterisk clear of the ascenders.
\newcommand{\mediant}{\,\raisebox{0.25em}{\small\color{mutedgray}*}\,}

% Psalm verses environment and command. Verses breathe a little deeper than
% prose, as psalm-verses does online.
\newenvironment{psalmverses}{%
  \setlength{\parindent}{0pt}%
  \setlength{\parskip}{2pt}%
  \setstretch{1.35}%
}{%
  \par%
}

% Hymn stanza environment: deeper verse leading, as hymn-verses does. A
% stanza is one paragraph, so the page-flow penalties keep it whole unless it
% is longer than a page.
\newenvironment{hymnverses}{%
  \setstretch{1.35}%
  \interlinepenalty=10000%
}{%
  \par%
}

% \psalmverse{num}{text} — num is empty for unnumbered first verse.
% Numbered verses hang from a muted right-aligned gutter so every spoken line
% shares one left edge, as verse-num does on the web hour.
\newcommand{\psalmverse}[2]{%
  \if\relax\detokenize{#1}\relax%
    \noindent #2\par\capclear%
  \else%
    \noindent\hangindent=2em\hangafter=1\makebox[1.4em][r]{{\small\color{mutedgray}#1}}\hspace{0.6em}#2\par%
  \fi%
}

% Gloria Patri: two verses, each pointed at its own mediant, on their own
% lines, standing a little apart from the verses above as its own stanza.
\newcommand{\gloriapatri}[2]{\medskip\noindent #1\\#2\par}

% Spoken-role sigils set in rubric red, as the web sigil column does.
% \Vbar/\Rbar come from gregoriotex, which the preamble loads above.
\newcommand{\Vsig}{{\color{rubricred}\Vbar{}}}
\newcommand{\Rsig}{{\color{rubricred}\Rbar{}}}
\newcommand{\allsig}{{\color{rubricred}\scshape All:}\kern0.3em{}}
\newcommand{\blessingsig}{{\color{rubricred}Blessing.}\kern0.3em{}}

% ℣ and ℟ lines: the sigil sits in a narrow gutter and a wrapped remainder
% tucks under the text, not the sigil, as the web sigil column does.
\newlength{\sigilwidth}
\setlength{\sigilwidth}{1.3em}
\newcommand{\sigilline}[2]{%
  \noindent\hangindent=\sigilwidth\hangafter=1\makebox[\sigilwidth][l]{#1}\ignorespaces#2\par%
}
\newcommand{\versicle}[1]{\sigilline{\Vsig}{#1}}
\newcommand{\response}[1]{\sigilline{\Rsig}{#1}}

% Antiphon: a red small-cap sigil with the body in ordinary roman, hanging so
% a wrapped remainder tucks under its own line, as body antiphons do online.
\newcommand{\ant}[1]{\noindent\hangindent=1.35em\hangafter=1{\color{rubricred}\scshape Ant.}\hspace{0.3em}#1\par}
% Opening acclamation (not an antiphon)
\newcommand{\acclamation}[1]{\noindent #1\par}
% Initial response of a Short Responsory. No \par here — call sites append the
% rest of the response (and a single trailing \par) so lettrine can wrap line 2.
\newcommand{\shortresponse}[2]{\dropcap{#1}{#2}}
